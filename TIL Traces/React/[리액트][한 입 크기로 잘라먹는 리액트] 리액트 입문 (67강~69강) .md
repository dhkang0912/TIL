  const onUpdate = (targetId) => {
    // todos State의 값들 중에 tagetId와 일치하는 id를 갖는 todo Item의 isDone 변경
    setTodos(
      <!-- map 메소드를 통해서 todos 배열에 접근하여 변경해줌 -->

        // 만약 targetId와 id가 일치하지 않는 todo라면 isDone을 변경하지 않고 기존 todo를 리턴

  const onDelete = (targetId)=>{

    // => targetId와 같지 않은 todo로만 배열을 만들어서 반환
    setTodos(todos.filter((todo)=>todo.id !== targetId))
  }

  return (
    <div className="App">
      <Header />
      <Editor onCreate={onCreate} />
      <List todos={todos} onUpdate={onUpdate} onDelete={onDelete}/>
    </div>
  );
}

export default App;

- List.jsx

import "./List.css";
import TodoItem from "./TodoItem";
import { useState } from "react";

const List = ({ todos, onUpdate, onDelete }) => {
  <!-- 검색어를 저장하는 State가 필요, 검색 인풋에 따라 실시간으로 업데이트 되어야 함 -->
  const [search, setSearch] = useState("");

  <!-- 입력값을 받아서 search State를 실시간으로 변경해줌 -->
  const onChangeSearch = (e) => {
    setSearch(e.target.value);
  };

  <!-- search State를 통해 검색한 값만 남기고 필터된 값을 기반으로 TodoItem 컴포넌트로 넘기기 -->
  const getFilteredData = () => {
    if (search === "") {
      // 검색어가 없다면 전체 todos를 반환
      return todos;
    }

    return todos.filter((todo) =>
      // 배열의 모든 todos를 순회하면서 각 todo의 content에 {search}에 들어있는 단어가 포함되면 true를 반환하고 그 값들만 배열에 모아서 반환함
      // 소문자로 전환한 검색어가 소문자로 전환한 내용에 포함되어있는 것들을 배열로 반환 <= 소문자로 검색해도 검색 가능하게 하는 법
      todo.content.toLowerCase().includes(search.toLowerCase())
    );
  };

  const filteredTodos = getFilteredData();

  return (
    <div className="List">
      <h4>Todo List✨</h4>
      <input
        value={search}
        onChange={onChangeSearch}
        placeholder="검색어를 입력하세요"
      />
      <div className="todos_wrapper">
      <!-- TodoItem에서 업데이트, 삭제가 가능하니 해당 state와 함수를 props로 넘겨줘야 함 -->
        {filteredTodos.map((todo) => {
          return <TodoItem key={todo.id} {...todo} onUpdate={onUpdate} onDelete={onDelete} />;
        })}
      </div>
    </div>
  );
};

export default List;
